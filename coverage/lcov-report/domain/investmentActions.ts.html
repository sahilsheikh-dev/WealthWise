import { NotFoundError } from './actions';
import { findLinkedTransaction, investCategoryFor } from './links';
import type { AppData, Holding, InvestmentEntry, Transaction } from './types';
export type InvestmentEntryInput = Omit<InvestmentEntry, 'id'>;
export type InvestmentEntryPatch = Partial<Pick<InvestmentEntry, 'date' | 'amount' | 'note'>>;
function getHolding(data: AppData, id: string): Holding { const item = data.holdings.find((holding) => holding.id === id); if (!item) throw new NotFoundError('Holding', id); return item; }
function text(holding: Holding, kind: InvestmentEntry['kind']): string { return `${holding.name} ${kind === 'Withdrawal' ? 'withdrawal' : holding.kind === 'plan' ? 'installment' : 'deposit'}`; }
function transaction(holding: Holding, entry: InvestmentEntry, id: string): Transaction { return { id, type: entry.kind === 'Withdrawal' ? 'income' : 'invest', amount: entry.amount, date: entry.date, category: entry.kind === 'Withdrawal' ? 'Withdrawal' : investCategoryFor(holding.type), note: text(holding, entry.kind), source: { kind: 'holding-entry', parentId: holding.id, entryId: entry.id } }; }
function currentValue(holding: Holding, date: string, excluding?: string): number { const base = holding.kind === 'manual' ? holding.value : holding.opening; return Math.max(0, base + (holding.logs ?? []).filter((entry) => entry.id !== excluding && entry.date <= date).reduce((sum, entry) => sum + (entry.kind === 'Withdrawal' ? -entry.amount : entry.amount), 0)); }
export function addInvestmentEntry(data: AppData, holdingId: string, input: InvestmentEntryInput, newId: () => string): AppData { const holding = getHolding(data, holdingId); const entry = { ...input, id: newId() }; if (entry.amount <= 0) throw new Error('Amount must be greater than zero'); if (entry.kind === 'Withdrawal' && entry.amount > currentValue(holding, entry.date)) throw new Error(`Exceeds the current value of ₹${currentValue(holding, entry.date)}`); const updated = { ...holding, logs: [...(holding.logs ?? []), entry] } as Holding; return { ...data, holdings: data.holdings.map((item) => item.id === holdingId ? updated : item), transactions: [...data.transactions, transaction(holding, entry, newId())] }; }
export function updateInvestmentEntry(data: AppData, holdingId: string, entryId: string, patch: InvestmentEntryPatch): AppData { const holding = getHolding(data, holdingId); const current = (holding.logs ?? []).find((entry) => entry.id === entryId); if (!current) throw new NotFoundError('Investment entry', entryId); const entry = { ...current, ...patch, kind: current.kind }; if (entry.kind === 'Withdrawal' && entry.amount > currentValue(holding, entry.date, entryId)) throw new Error(`Exceeds the current value of ₹${currentValue(holding, entry.date, entryId)}`); const updated = { ...holding, logs: (holding.logs ?? []).map((item) => item.id === entryId ? entry : item) } as Holding; const linked = findLinkedTransaction(data, 'holding-entry', entryId); return { ...data, holdings: data.holdings.map((item) => item.id === holdingId ? updated : item), transactions: linked ? data.transactions.map((item) => item.id === linked.id ? { ...item, amount: entry.amount, date: entry.date } : item) : data.transactions }; }
export function deleteInvestmentEntry(data: AppData, holdingId: string, entryId: string): AppData { const holding = getHolding(data, holdingId); if (!(holding.logs ?? []).some((entry) => entry.id === entryId)) throw new NotFoundError('Investment entry', entryId); return { ...data, holdings: data.holdings.map((item) => item.id === holdingId ? { ...holding, logs: (holding.logs ?? []).filter((entry) => entry.id !== entryId) } : item), transactions: data.transactions.filter((item) => !(item.source?.kind === 'holding-entry' && item.source.entryId === entryId)) }; }
